문자와 문자열 (Character & String)

💬문자열이란?

char 타입 데이터로 구성된 1차원 배열로, 반드시 NULL 문자 '\0'으로 끝나야 합니다.

단일 따옴표, 1 byte만 사용

C
#include <stdio.h>
#include <string.h>   // strlen() 사용

int main(void)
{
  char str[6] = {'H','E','L','L','O','\0'};
  int  i = 0;

  /* 방법 1: NULL 문자까지 문자 단위 출력 */
  while (str[i] != '\0') { printf("%c", str[i]); i++; }
  printf("\n");

  /* 방법 2: %s 형식 지정자 */
  printf("%s\n", str);

  printf("sizeof(str) = %zu\n", sizeof(str));  // 6
  printf("strlen(str) = %zu\n", strlen(str));  // 5
  return 0;
}

문자 배열의 초기화 (Initialization)

문자 배열 초기화 패턴 — 직접 확인해보세요

char str[4] = {'a','b','c','\0'} a b c \0 4 명시적 초기화
char str[4] = "abc" a b c \0 4 문자열 리터럴

문자 입출력 라이브러리 함수 (stdio.h)

⌨️getchar() — 문자 입력

int getchar(void);
// 키보드에서 문자 하나 읽기
// Enter 후 전달 (버퍼 방식)
// 리턴: 읽은 문자 (int) 또는 EOF
// ≈ scanf("%c", &c)

반환 타입이 int인 이유: EOF(-1)를 검사하기 위함

🖥️putchar() — 문자 출력

void putchar(int c);
// 문자 c를 화면에 출력
// ≈ printf("%c", c)

printf/scanf의 경량(lightweight) 문자 전용 버전

LAB: p8.c — EOF까지 문자 에코
C
#include <stdio.h>
int main(void)
{
  int ch;
  /* EOF: Linux=Ctrl+D, Windows=Ctrl+Z */
  while ((ch = getchar()) != EOF)
    putchar(ch);
  return 0;
}
💡
ch를 int로 선언하는 이유: char는 -128~127 또는 0~255이지만, EOF는 -1이므로 int가 필요합니다.

LAB: p10_1.c — 대소문자 토글 + 단어 카운터
C
/* 대소문자 반전 출력 */
#include <stdio.h>
#include <ctype.h>
int main(void) {
  int c;
  while ((c = getchar()) != EOF) {
    if (islower(c)) c = toupper(c);
    else             c = tolower(c);
    putchar(c);
  }
  return 0;
}
// 입력: "Hello World"  →  출력: "hELLO wORLD"

/* 단어 개수 세기 — isspace() 활용 */
int word_cnt(const char *s) {
  int w_cnt = 0;
  while (*s != '\0') {
    while (isspace(*s)) ++s;      // 공백 건너뜀
    if (*s != '\0') {
      ++w_cnt;
      while (!isspace(*s) && *s != '\0') ++s; // 단어 건너뜀
    }
  }
  return w_cnt;
}
// "a big apple" → 3

📌
모든 함수 사용 전 #include <string.h> 필수. 포인터 매개변수의 NULL 여부는 자동 체크하지 않으므로 호출 전 직접 확인이 필요합니다.

📏strlen(s) — 문자열 길이

'\0'이 나올 때까지 문자 개수를 카운팅. '\0' 자체는 포함하지 않음.

strlen 내부 구현C
size_t strlen(const char *s)
{
  size_t len;
  for (len = 0; *s != '\0'; ++s)
    ++len;
  return len;
}

📋strcpy(dest, src) — 문자열 복사

src 내용을 dest에 복사. '\0'도 함께 복사됩니다. dest 공간이 충분해야 함!

strcpy 내부 구현C
char *strcpy(char *dst, const char *src)
{
  char *p = dst;
  while ((*p++ = *src++) != '\0'); // '\0'도 복사!
  return dst;
}
// strncpy(dst, src, n): 최대 n글자만 복사 (더 안전)
⚠️
dst 배열 크기 ≥ strlen(src) + 1을 반드시 확인하세요! 더 안전한 strncpy(dst, src, sizeof(dst)-1) 사용을 권장합니다.

🔗strcat(s1, s2) — 문자열 연결

s1의 끝 '\0' 위치에 s2를 이어 붙입니다. s1의 공간이 충분해야 합니다.

strcat 내부 구현 v1C
char *strcat(char *dst, const char *src)
{
  char *p = dst;
  while (*p != '\0') p++; // '\0' 찾기
  while ((*p++ = *src++) != '\0');
  return dst;
}
strcat 내부 구현 v2C
char *strcat(char *dst, const char *src)
{
  int len = strlen(dst);
  strcpy(&dst[len], src);
  return dst;
}

⚖️strcmp(s1, s2) — 문자열 비교

두 문자열을 사전 순서로 비교합니다.

  • s1 == s2 → 0 반환
  • s1 < s2 → 음수 반환
  • s1 > s2 → 양수 반환
strcmp 내부 구현C
int strcmp(const char *s1, const char *s2)
{
  while(*s1 == *s2) {
    if (*s1 == '\0') return 0;  // 완전히 같음
    s1++; s2++;
  }
  return (int)((unsigned char)*s1 - (unsigned char)*s2);
}

▶ 비교를 눌러보세요
⚠️
"Apple" vs "apple"? ASCII에서 대문자 A(65) < 소문자 a(97)이므로 strcmp("Apple","apple")은 음수를 반환합니다 — "Apple"이 사전적으로 작습니다.

🔎strchr / strrchr

// 문자 c 첫 번째 위치 주소 반환
char *strchr(const char *s, int c);

// 문자 c 마지막 위치 주소 반환
char *strrchr(const char *s, int c);

// 없으면 NULL 반환

// pos → "llo"

🔍strstr

// 문자열 s1에서 패턴 s2의
// 첫 번째 위치 주소 반환
char *strstr(const char *s1,
             const char *s2);

char *p = strstr("beautiful sky",
                 "sky");
// p → "sky"

✂️strtok(s, delim) — 토큰 분리

문자열 s를 구분자 delim으로 분리하여 개별 토큰으로 나눕니다.

  • 처음 호출: strtok(s, delim)
  • 이후 호출: strtok(NULL, delim) — 이전 위치부터 계속
  • 원본 문자열을 수정하므로 const 문자열에 사용 불가!
strtok 사용 예시
C
#include <stdio.h>
#include <string.h>

int main(void) {
  char str[] = "Hello,World,2025";
  char *token;

  token = strtok(str, ",");   // "Hello"
  while (token != NULL) {
    printf("%s\n", token);
    token = strtok(NULL, ","); // 다음 토큰
  }
  return 0;
}

메모리 버퍼 입출력 — sprintf / sscanf

💾sprintf / sscanf란?

printf/scanf와 동일한 기능이지만 입출력 대상이 화면/키보드가 아닌 메모리 버퍼(문자열)입니다.

  • sprintf(buf, fmt, ...) — 형식화된 문자열을 buf에 출력
  • sscanf(buf, fmt, ...) — buf에서 형식에 맞게 값을 읽기
  • buf의 공간이 충분해야 함! (버퍼 오버플로우 주의)
LAB: p18.c — 날짜 포맷 변환
C
#include <stdio.h>

int main(void) {
  int  year=2025, month=10, day=6;
  char date[16];

  /* sprintf: 숫자 → 문자열 버퍼 */
  sprintf(date, "%4d-%02d-%02d", year, month, day);
  printf("date = %s\n", date); // date = 2025-10-06

  /* sscanf: 문자열 버퍼 → 숫자 */
  sscanf(date, "%d-%d-%d", &year, &month, &day);
  printf("%4d-%02d-%02d\n", year, month, day); // 2025-10-06
  return 0;
}
sprintf 인터랙티브 포맷터
▶ 실행을 눌러보세요
💡
sprintf/sscanf 주요 활용 • 숫자 → 문자열 변환 (예: 날짜 포맷팅, 파일명 생성)
• 문자열 → 여러 변수 파싱 (예: CSV 파일 한 줄 처리)
• 로그 메시지 버퍼에 포맷화해서 저장 후 한 번에 출력

ASCII 코드 기준: 'A'=65, 'a'=97입니다.
첫 글자 비교: 'A'(65) < 'a'(97) → 'A' - 'a' = 음수 반환.
따라서 "Apple"이 사전적으로 "apple"보다 작습니다.

char str[4] = "abcde"; 선언 후 printf("%s", str)의 결과는?
A "abcde" 출력
B "abc" 출력 (안전하게 잘림)
C 컴파일 에러
D 정의되지 않은 동작 (Undefined Behavior)
✅ 정답: D — 정의되지 않은 동작 (UB)
char str[4] = "abcde": 배열 크기(4)보다 문자가 많아 '\0'이 저장되지 않습니다. str = {'a','b','c','d'}
printf("%s")는 '\0'이 나올 때까지 메모리를 계속 읽으므로 배열 경계를 넘어 쓰레기 값을 출력하거나 세그폴트가 발생합니다 (Undefined Behavior).
QUIZ 05 / 06
다음 코드 실행 후 s1의 내용은?
char s1[30] = "beautiful ";
char s2[] = "day";
strcat(s1, s2);
strcat(s1, "!");
A "beautiful "
B "beautiful day"
C "beautiful day!"
D "day!"
✅ 정답: C — "beautiful day!"
1단계: strcat(s1, s2) → s1 = "beautiful day" (s2의 '\0' 자리에 이어붙임)
2단계: strcat(s1, "!") → s1 = "beautiful day!"
s1의 배열 크기(30)가 최종 문자열 길이(14+'\0'=15)보다 크므로 안전합니다.
QUIZ 06 / 06
getchar()의 반환 타입이 char가 아닌 int인 이유는?
A char는 반환 타입으로 사용할 수 없어서
B EOF(-1)를 구분하기 위해 char 범위보다 큰 타입이 필요해서
C ASCII 코드가 항상 정수이기 때문에
D 성능 최적화를 위해
✅ 정답: B
char는 보통 -128~127 (signed) 또는 0~255 (unsigned) 범위입니다.
EOF는 -1로 정의되어 있는데, unsigned char로 받으면 -1이 255로 해석되어 EOF를 정상 문자와 구분할 수 없습니다.
int를 사용하면 0~255의 문자 값과 EOF(-1)를 모두 올바르게 구분할 수 있습니다.
